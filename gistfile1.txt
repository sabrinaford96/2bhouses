    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, doc, onSnapshot, updateDoc, setDoc, getDoc, collection, addDoc, deleteDoc, increment } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";
        import { setLogLevel } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // Set Firebase debug log level
        setLogLevel('debug');

        // Global variables for Firebase configuration, provided by the environment
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
        const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : {};
        const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? initialAuthToken : null;

        // --- Firebase Initialization ---
        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);
        
        // --- DOM Elements ---
        const dashboardEl = document.getElementById('dashboard');
        const loadingEl = document.getElementById('loading');
        const userIdDisplayEl = document.getElementById('user-id-display');
        const eventLogEl = document.getElementById('event-log');
        const memberModalEl = document.getElementById('member-modal');
        const pointsModalEl = document.getElementById('points-modal');
        const memberModalTitleEl = document.getElementById('modal-title');
        const membersListEl = document.getElementById('members-list');
        const newMemberInputEl = document.getElementById('new-member-name');
        const awardMemberNameEl = document.getElementById('award-member-name');
        const pointsAmountEl = document.getElementById('points-amount');
        const pointsCategoryEl = document.getElementById('points-category');

        let currentHouseName = '';
        let currentMemberId = '';

        const houses = [
            { name: "Earth", color: "green-500", icon: "🌎" },
            { name: "Air", color: "blue-300", icon: "🌬️" },
            { name: "Fire", color: "red-500", icon: "🔥" },
            { name: "Water", color: "blue-500", icon: "🌊" }
        ];

        // Function to create a house card element
        function createHouseCard(house) {
            const card = document.createElement('div');
            card.className = `house-card relative overflow-hidden bg-${house.color} bg-opacity-10 p-6 rounded-3xl shadow-lg flex flex-col items-center text-center transition-all duration-300 transform hover:scale-105`;
            card.onclick = () => showMembers(house.name);
            card.innerHTML = `
                <div class="text-4xl mb-2">${house.icon}</div>
                <h3 class="text-2xl font-bold text-gray-800 mb-1">${house.name}</h3>
                <div class="text-6xl sm:text-7xl font-extrabold text-${house.color} score-display" id="score-${house.name.replace(/\s/g, '-')}">0</div>
            `;
            return card;
        }
        
        // Log an event to the UI
        function logEvent(message) {
            const li = document.createElement('li');
            li.className = 'log-item text-sm text-gray-500 p-2 border-b last:border-b-0 border-gray-100';
            li.textContent = message;
            eventLogEl.prepend(li);
            if (eventLogEl.children.length > 10) {
                eventLogEl.removeChild(eventLogEl.lastChild);
            }
        }

        // --- Firebase Auth ---
        let userId = null;
        onAuthStateChanged(auth, async (user) => {
            if (user) {
                userId = user.uid;
                userIdDisplayEl.textContent = `Your User ID: ${userId}`;
                dashboardEl.classList.remove('hidden');
                loadingEl.classList.add('hidden');
                
                // Set up the listeners after auth is ready
                setupListeners();
            } else {
                // If the initial token is not present, sign in anonymously.
                try {
                    if (initialAuthToken) {
                        await signInWithCustomToken(auth, initialAuthToken);
                    } else {
                        await signInAnonymously(auth);
                    }
                } catch (error) {
                    console.error("Error signing in:", error);
                    loadingEl.textContent = "Error loading app. Please check your network.";
                }
            }
        });

        // --- Firestore Operations ---
        
        function showMembers(houseName) {
            currentHouseName = houseName;
            memberModalTitleEl.textContent = `${houseName} Members`;
            memberModalEl.classList.remove('hidden');
            membersListEl.innerHTML = '<div class="text-center text-gray-400">Loading members...</div>';
            
            const membersCollectionRef = collection(db, `/artifacts/${appId}/public/data/house_points/${houseName.replace(/\s/g, '-')}/members`);
            
            // Listen for real-time member updates
            onSnapshot(membersCollectionRef, (snapshot) => {
                membersListEl.innerHTML = '';
                if (snapshot.empty) {
                    membersListEl.innerHTML = '<div class="text-center text-gray-400 p-4">No members yet. Add the first one!</div>';
                } else {
                    snapshot.forEach(memberDoc => {
                        const memberData = memberDoc.data();
                        const memberItem = document.createElement('div');
                        memberItem.className = 'flex justify-between items-center p-2 rounded-lg hover:bg-gray-100 transition-colors duration-200';
                        memberItem.innerHTML = `
                            <span>${memberData.name}</span>
                            <div class="space-x-2">
                                <button onclick="showPointsModal('${memberDoc.id}', '${memberData.name}', '${houseName}')" class="bg-blue-500 hover:bg-blue-600 text-white text-sm font-bold py-1 px-3 rounded-full transition-colors">Award Points</button>
                                <button onclick="deleteMember('${houseName.replace(/\s/g, '-')}', '${memberDoc.id}')" class="text-red-500 hover:text-red-700 transition-colors duration-200 font-bold">&times;</button>
                            </div>
                        `;
                        membersListEl.appendChild(memberItem);
                    });
                }
            });
        }
        
        async function addMember() {
            const name = newMemberInputEl.value.trim();
            if (!name) {
                newMemberInputEl.placeholder = "Name cannot be empty!";
                newMemberInputEl.classList.add('border-red-500');
                setTimeout(() => {
                    newMemberInputEl.classList.remove('border-red-500');
                    newMemberInputEl.placeholder = "Enter new member's name";
                }, 1500);
                return;
            }

            const membersCollectionRef = collection(db, `/artifacts/${appId}/public/data/house_points/${currentHouseName.replace(/\s/g, '-')}/members`);

            try {
                await addDoc(membersCollectionRef, {
                    name: name,
                    addedBy: userId,
                    timestamp: new Date().toISOString()
                });
                newMemberInputEl.value = '';
                logEvent(`New member '${name}' added to ${currentHouseName}`);
            } catch (e) {
                console.error("Error adding member: ", e);
                logEvent(`Error adding member '${name}'`);
            }
        }

        async function deleteMember(houseName, memberId) {
            const memberDocRef = doc(db, `/artifacts/${appId}/public/data/house_points/${houseName}/members/${memberId}`);
            try {
                await deleteDoc(memberDocRef);
                logEvent(`Member removed from ${houseName.replace(/-/g, ' ')}`);
            } catch (e) {
                console.error("Error deleting member: ", e);
                logEvent(`Error removing member`);
            }
        }
        
        function showPointsModal(memberId, memberName, houseName) {
            currentMemberId = memberId;
            currentHouseName = houseName;
            awardMemberNameEl.textContent = memberName;
            pointsModalEl.classList.remove('hidden');
        }

        function closePointsModal() {
            pointsModalEl.classList.add('hidden');
        }

        async function submitPoints() {
            const points = parseInt(pointsAmountEl.value, 10);
            const category = pointsCategoryEl.value;
            const memberName = awardMemberNameEl.textContent;

            if (isNaN(points) || points <= 0) {
                pointsAmountEl.classList.add('border-red-500');
                setTimeout(() => pointsAmountEl.classList.remove('border-red-500'), 1500);
                return;
            }

            // Atomically update the house's total points
            const houseDocRef = doc(db, `/artifacts/${appId}/public/data/house_points/${currentHouseName.replace(/\s/g, '-')}`);
            
            try {
                // Use setDoc with merge: true to create the document if it doesn't exist
                await setDoc(houseDocRef, {
                    points: increment(points),
                    lastUpdatedBy: userId,
                    timestamp: new Date().toISOString()
                }, { merge: true });
                
                // Add a new document to the activities subcollection
                const activitiesCollectionRef = collection(db, `/artifacts/${appId}/public/data/house_points/${currentHouseName.replace(/\s/g, '-')}/activities`);
                await addDoc(activitiesCollectionRef, {
                    memberId: currentMemberId,
                    memberName: memberName,
                    points: points,
                    category: category,
                    addedBy: userId,
                    timestamp: new Date().toISOString()
                });

                logEvent(`${memberName} received ${points} points for ${category} for ${currentHouseName}.`);
                
                closePointsModal();
            } catch (e) {
                console.error("Error awarding points: ", e);
                logEvent(`Error awarding points to ${memberName}`);
            }
        }

        function closeModal() {
            memberModalEl.classList.add('hidden');
        }

        // --- Real-time Listener ---
        function setupListeners() {
            houses.forEach(house => {
                const docPath = `/artifacts/${appId}/public/data/house_points/${house.name.replace(/\s/g, '-')}`;
                const houseDocRef = doc(db, docPath);

                // Set up initial dashboard structure
                const houseCard = createHouseCard(house);
                dashboardEl.appendChild(houseCard);
                const scoreDisplay = document.getElementById(`score-${house.name.replace(/\s/g, '-')}`);

                // Listen for real-time updates
                onSnapshot(houseDocRef, (docSnap) => {
                    const newPoints = docSnap.exists() ? docSnap.data().points : 0;
                    
                    if (scoreDisplay.textContent !== newPoints.toString()) {
                        scoreDisplay.textContent = newPoints;
                        scoreDisplay.classList.add('animate-pop');
                        
                        setTimeout(() => {
                            scoreDisplay.classList.remove('animate-pop');
                        }, 300);
                    }
                }, (error) => {
                    console.error("Error listening to document: ", error);
                    logEvent(`Error with live updates for ${house.name}`);
                });
            });
        }
        
        // Expose functions to the global scope for onclick attributes
        window.showMembers = showMembers;
        window.addMember = addMember;
        window.deleteMember = deleteMember;
        window.closeModal = closeModal;
        window.showPointsModal = showPointsModal;
        window.closePointsModal = closePointsModal;
        window.submitPoints = submitPoints;

    </script>
